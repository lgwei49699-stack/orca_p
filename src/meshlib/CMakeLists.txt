elseif(WIN32)

        message(FATAL_ERROR "The integrated MeshLib Windows evaluation supports x64 only.")

    # Official non-IteratorDebug archives use iterator level 0 and the DLL CRT.
    # Do not silently change Orca's STL ABI or map its Debug build to this SDK.

        set(_configurations "${CMAKE_BUILD_TYPE}")
    endif()
    if(NOT _configurations)
        message(FATAL_ERROR "MeshLib Windows evaluation requires an explicit Release, RelWithDebInfo, or MinSizeRel configuration.")
    endif()
    set(_host_flags "")
    foreach(_configuration IN LISTS _configurations)
        if(NOT _configuration MATCHES "^(Release|RelWithDebInfo|MinSizeRel)$")
            message(FATAL_ERROR "MeshLib Windows evaluation supports Release, RelWithDebInfo, or MinSizeRel only. Configure CMAKE_CONFIGURATION_TYPES=Release for a multi-config generator; Debug requires a separately matched SDK/STL ABI.")
        endif()
        string(TOUPPER "${_configuration}" _upper_configuration)
        string(APPEND _host_flags " ${CMAKE_CXX_FLAGS_${_upper_configuration}}")
    endforeach()
    string(APPEND _host_flags " ${CMAKE_CXX_FLAGS}")
    if(_host_flags MATCHES "(^|[ ;])[-/](MTd?|MDd)([ ;]|$)" OR
       _host_flags MATCHES "_ITERATOR_DEBUG_LEVEL[= ]+[1-9]")
        message(FATAL_ERROR "MeshLib Windows evaluation requires /MD and _ITERATOR_DEBUG_LEVEL=0 throughout Orca; /MT, /MDd, or a debug STL ABI is incompatible.")
    endif()
    if(DEFINED CMAKE_MSVC_RUNTIME_LIBRARY AND NOT CMAKE_MSVC_RUNTIME_LIBRARY STREQUAL "" AND
       NOT CMAKE_MSVC_RUNTIME_LIBRARY STREQUAL "MultiThreadedDLL" AND
       NOT CMAKE_MSVC_RUNTIME_LIBRARY STREQUAL "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")

    endif()

if(WIN32)
    list(APPEND _prefixes "${ORCA_MESHLIB_SDK_ROOT}/install")
    set(_include_suffix "include")
elseif(APPLE)
    set(_include_suffix "include")

    list(APPEND _prefixes "${ORCA_MESHLIB_SDK_ROOT}/usr/local")
    set(_include_suffix "include/MeshLib")

    if(EXISTS "${_prefix}/${_include_suffix}/MRMesh/MRMesh.h")
        set(_sdk_prefix "${_prefix}")
        set(_sdk_include "${_prefix}/${_include_suffix}")
        break()
    endif()
endforeach()
if(NOT _sdk_prefix)
    message(FATAL_ERROR "The selected SDK is missing ${_include_suffix}/MRMesh/MRMesh.h; choose a complete fixed-version SDK prefix.")
endif()

if(WIN32)
    set(_version_file "${_sdk_prefix}/app/Release/mr.version")
    if(NOT EXISTS "${_version_file}")
        message(FATAL_ERROR "The Windows SDK is missing app/Release/mr.version; use the complete official non-IteratorDebug archive.")
    endif()
    file(READ "${_version_file}" _sdk_version LIMIT 128)
    string(STRIP "${_sdk_version}" _sdk_version)
    if(NOT _sdk_version STREQUAL "3.1.3.429")
        message(FATAL_ERROR "Unsupported MeshLib SDK version: ${_sdk_version}; expected 3.1.3.429.")
    endif()
    set(_runtime_dir "${_sdk_prefix}/app/Release")
    set(_mesh_runtime "${_runtime_dir}/MRMesh.dll")
    set(_voxels_runtime "${_runtime_dir}/MRVoxels.dll")
    set(_mesh_import "${_sdk_prefix}/lib/Release/MRMesh.lib")
    set(_voxels_import "${_sdk_prefix}/lib/Release/MRVoxels.lib")
    foreach(_import IN ITEMS _mesh_import _voxels_import)
        if(NOT EXISTS "${${_import}}")
            message(FATAL_ERROR "The Windows SDK import library is missing: ${${_import}}")
        endif()
    endforeach()
else()
    set(_version_file "${_sdk_include}/MRMesh/config_cmake.h")
    if(NOT EXISTS "${_version_file}")
        message(FATAL_ERROR "The SDK is missing MRMesh/config_cmake.h; a complete versioned C++ SDK is required.")
    endif()
    file(STRINGS "${_version_file}" _version_line REGEX "^#define MESHLIB_VERSION ")
    if(NOT _version_line STREQUAL "#define MESHLIB_VERSION \"v3.1.3.429\"")
        message(FATAL_ERROR "Unsupported MeshLib SDK header version: ${_version_line}")
    endif()

        set(_runtime_suffix ".dylib")

    endif()

    set(_voxels_runtime "${_runtime_dir}/libMRVoxels${_runtime_suffix}")

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # Keep Orca's old static runtime symbols out of the executable's dynamic
    # table, where they could interpose on the SDK's newer shared dependencies.
    # This is intentionally scoped to known dependency archives, not ALL Orca
    # symbols. It does not replace a native readelf/binding/runtime regression.
    set(_private_dependencies TBB::tbb TBB::tbbmalloc OpenVDB::openvdb OpenSSL::Crypto OpenSSL::SSL
        ZLIB::ZLIB Blosc::blosc Boost::iostreams Boost::filesystem Boost::system Boost::thread
        Boost::log Boost::log_setup Boost::regex Boost::date_time Boost::chrono Boost::atomic)
    foreach(_dependency IN LISTS _private_dependencies)
        if(NOT TARGET "${_dependency}")
            continue()
        endif()
        get_target_property(_kind "${_dependency}" TYPE)
        if(_kind STREQUAL "SHARED_LIBRARY" OR _kind STREQUAL "MODULE_LIBRARY")
            message(FATAL_ERROR "MeshLib Linux evaluation requires static Orca dependency ${_dependency}; mixed shared runtimes are not supported.")
        endif()
        if(_kind STREQUAL "INTERFACE_LIBRARY")
            continue()
        endif()
        get_target_property(_imported_configs "${_dependency}" IMPORTED_CONFIGURATIONS)
        set(_location_properties IMPORTED_LOCATION)
        foreach(_config IN LISTS _imported_configs)
            string(TOUPPER "${_config}" _config)
            list(APPEND _location_properties "IMPORTED_LOCATION_${_config}")
        endforeach()
        foreach(_property IN LISTS _location_properties)
            get_target_property(_location "${_dependency}" "${_property}")
            if(_location AND NOT _location MATCHES "\\.a$")
                message(FATAL_ERROR "MeshLib Linux evaluation requires static Orca dependency ${_dependency}, found ${_location}.")
            endif()
        endforeach()
        target_link_options(orca_meshlib_adapter INTERFACE "LINKER:--exclude-libs,$<TARGET_FILE_NAME:${_dependency}>")
    endforeach()
endif()
