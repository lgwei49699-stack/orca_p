cmake_minimum_required(VERSION 3.18)
project(OrcaMeshLibEvaluationWorker LANGUAGES CXX)

# Deliberately a separate, opt-in build. No SDK is downloaded, installed, copied
# into Orca, or included in an application package by this project.
option(ORCA_MESHLIB_LOCAL_EVALUATION "Build the separately licensed MeshLib worker for local evaluation" OFF)
if(NOT ORCA_MESHLIB_LOCAL_EVALUATION)
    message(FATAL_ERROR "Enable ORCA_MESHLIB_LOCAL_EVALUATION only for an authorized local SDK evaluation. Commercial distribution needs separate authorization.")
endif()
set(ORCA_MESHLIB_SDK_ROOT "" CACHE PATH "Fixed MeshLib 3.1.3.429 SDK prefix containing include and lib")
set(ORCA_MESHLIB_EXTRA_INCLUDE_DIRS "" CACHE STRING "Explicit third-party SDK header directories, if not included in the SDK")
set(ORCA_MESHLIB_LIBRARY_DIR "${ORCA_MESHLIB_SDK_ROOT}/lib" CACHE PATH "SDK native library directory")
if(NOT IS_ABSOLUTE "${ORCA_MESHLIB_SDK_ROOT}")
    message(FATAL_ERROR "ORCA_MESHLIB_SDK_ROOT must be an absolute path to the extracted, fixed-version SDK")
endif()
set(_version_header "${ORCA_MESHLIB_SDK_ROOT}/include/MRMesh/config_cmake.h")
if(NOT EXISTS "${_version_header}")
    message(FATAL_ERROR "The SDK is missing MRMesh/config_cmake.h; a complete versioned C++ SDK is required")
endif()
file(STRINGS "${_version_header}" _version_line REGEX "^#define MESHLIB_VERSION ")
if(NOT _version_line STREQUAL "#define MESHLIB_VERSION \"v3.1.3.429\"")
    message(FATAL_ERROR "Unsupported MeshLib SDK header version: ${_version_line}")
endif()

find_library(_meshlib_mesh NAMES MRMesh PATHS "${ORCA_MESHLIB_LIBRARY_DIR}" NO_DEFAULT_PATH REQUIRED)
find_library(_meshlib_voxels NAMES MRVoxels PATHS "${ORCA_MESHLIB_LIBRARY_DIR}" NO_DEFAULT_PATH REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
if(WIN32)
    set(_mesh_runtime_default "${ORCA_MESHLIB_LIBRARY_DIR}/MRMesh.dll")
    set(_voxels_runtime_default "${ORCA_MESHLIB_LIBRARY_DIR}/MRVoxels.dll")
else()
    set(_mesh_runtime_default "${_meshlib_mesh}")
    set(_voxels_runtime_default "${_meshlib_voxels}")
endif()
set(ORCA_MESHLIB_MESH_RUNTIME "${_mesh_runtime_default}" CACHE FILEPATH "Actual MRMesh dynamic library (not a Windows import library)")
set(ORCA_MESHLIB_VOXELS_RUNTIME "${_voxels_runtime_default}" CACHE FILEPATH "Actual MRVoxels dynamic library (not a Windows import library)")
foreach(_library IN ITEMS ORCA_MESHLIB_MESH_RUNTIME ORCA_MESHLIB_VOXELS_RUNTIME)
    if(NOT IS_ABSOLUTE "${${_library}}" OR NOT EXISTS "${${_library}}")
        message(FATAL_ERROR "${_library} must identify the fixed SDK dynamic library")
    endif()
endforeach()
get_filename_component(_mesh_runtime "${ORCA_MESHLIB_MESH_RUNTIME}" REALPATH)
get_filename_component(_voxels_runtime "${ORCA_MESHLIB_VOXELS_RUNTIME}" REALPATH)
file(SHA256 "${_mesh_runtime}" _mesh_sha256)
file(SHA256 "${_voxels_runtime}" _voxels_sha256)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_version_header}" "${_mesh_runtime}" "${_voxels_runtime}")
configure_file(runtime_identity.hpp.in runtime_identity.hpp @ONLY)
get_filename_component(_orca_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(orca_meshlib_worker main.cpp "${_orca_root}/src/libslic3r/ReconstructionProcess.cpp")
set_target_properties(orca_meshlib_worker PROPERTIES
    OUTPUT_NAME OrcaMeshLibWorker CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF MACOSX_BUNDLE FALSE
    BUILD_RPATH "${ORCA_MESHLIB_LIBRARY_DIR};${ORCA_MESHLIB_LIBRARY_DIR}/lib")
target_compile_definitions(orca_meshlib_worker PRIVATE MR_USE_CMAKE_CONFIGURE_FILE)
target_include_directories(orca_meshlib_worker PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${_orca_root}/src")
target_include_directories(orca_meshlib_worker SYSTEM PRIVATE
    "${_orca_root}/deps_src" "${ORCA_MESHLIB_SDK_ROOT}/include" "${ORCA_MESHLIB_SDK_ROOT}/include/include" ${ORCA_MESHLIB_EXTRA_INCLUDE_DIRS})
target_link_libraries(orca_meshlib_worker PRIVATE "${_meshlib_mesh}" "${_meshlib_voxels}" Threads::Threads OpenSSL::Crypto ${CMAKE_DL_LIBS})
if(WIN32)
    target_link_libraries(orca_meshlib_worker PRIVATE Psapi.lib)
    target_compile_definitions(orca_meshlib_worker PRIVATE _ITERATOR_DEBUG_LEVEL=0 MR_ITERATOR_DEBUG_LEVEL=0 NOMINMAX)
endif()
if(MSVC)
    target_compile_options(orca_meshlib_worker PRIVATE /W4 /utf-8 /bigobj)
else()
    target_compile_options(orca_meshlib_worker PRIVATE -Wall -Wextra -Wpedantic)
endif()

# Intentionally no install(), package(), or SDK-copy target.
